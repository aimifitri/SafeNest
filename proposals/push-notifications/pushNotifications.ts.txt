import Constants from "expo-constants";
import * as Device from "expo-device";
import * as Notifications from "expo-notifications";
import { useRouter } from "expo-router";
import { useEffect } from "react";
import { Platform } from "react-native";

import { supabase } from "../../lib/supabase";

export type PushResult = { ok: true; token: string } | { ok: false; reason: "simulator" | "denied" | "no_project" | "failed"; message: string };

/** Call once at startup. Notifications that arrive while the app is open show as a banner. */
export function configurePushHandler() {
  Notifications.setNotificationHandler({
    handleNotification: async () => ({ shouldShowBanner: true, shouldShowList: true, shouldPlaySound: false, shouldSetBadge: false }),
  });
}

/**
 * Ask permission (only when the person taps "Turn on pickup alerts"), get this phone's token and save it.
 * Never call it at startup: the system prompt should appear when the person knows why.
 */
export async function enablePickupAlerts(): Promise<PushResult> {
  if (!Device.isDevice) return { ok: false, reason: "simulator", message: "Push alerts need a real phone." };

  if (Platform.OS === "android") {
    await Notifications.setNotificationChannelAsync("pickup", {
      name: "Pickup alerts",
      importance: Notifications.AndroidImportance.HIGH,
    });
  }

  let { status } = await Notifications.getPermissionsAsync();
  if (status !== "granted") status = (await Notifications.requestPermissionsAsync()).status;
  if (status !== "granted") return { ok: false, reason: "denied", message: "Alerts are off. You can turn them on in the phone's settings." };

  const projectId = Constants.expoConfig?.extra?.eas?.projectId ?? Constants.easConfig?.projectId;
  if (!projectId) return { ok: false, reason: "no_project", message: "This build has no Expo project id, so alerts cannot be set up." };

  try {
    const { data: token } = await Notifications.getExpoPushTokenAsync({ projectId });
    const { error } = await supabase.rpc("register_push_token", { new_token: token, new_platform: Platform.OS === "ios" ? "ios" : "android" });
    if (error) return { ok: false, reason: "failed", message: "Could not save alerts. Nothing was changed. Please try again." };
    return { ok: true, token };
  } catch {
    return { ok: false, reason: "failed", message: "Could not set up alerts. Please try again." };
  }
}

/** Call on log out, before the session is cleared, so the next person on this phone is not sent the old account's alerts. */
export async function disablePickupAlerts(token: string | null): Promise<void> {
  if (!token) return;
  await supabase.rpc("unregister_push_token", { old_token: token });
}

/** Put once inside the root layout. Tapping an alert opens Today, where the details are. */
export function usePushTapNavigation() {
  const router = useRouter();
  useEffect(() => {
    const sub = Notifications.addNotificationResponseReceivedListener(() => {
      router.navigate("/");
    });
    return () => sub.remove();
  }, [router]);
}
